    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // Elements
            const urlBar = document.getElementById('url-bar');
            const goBtn = document.getElementById('go-btn');
            const webview = document.getElementById('webview');
            const backBtn = document.getElementById('back-btn');
            const forwardBtn = document.getElementById('forward-btn');
            const refreshBtn = document.getElementById('refresh-btn');
            const proxyToggle = document.getElementById('proxy-toggle');
            const loadingOverlay = document.getElementById('loading-overlay');
            const settingsBtn = document.getElementById('settings-btn');
            const settingsModal = document.getElementById('settings-modal');
            const closeSettings = document.getElementById('close-settings');
            const saveSettings = document.getElementById('save-settings');
            const proxyServerInput = document.getElementById('proxy-server-input');
            const enableJs = document.getElementById('enable-js');
            const newTabBtn = document.getElementById('new-tab-btn');
            const tabsContainer = document.getElementById('tabs-container');

            // State
            let proxyEnabled = true;
            let currentProxyServer = 'https://cors-anywhere.herokuapp.com/';
            let javascriptEnabled = true;
            let history = [];
            let historyIndex = -1;

            // Initialize
            loadSettings();

            // Event Listeners
            urlBar.addEventListener('keypress', function(e) {
                if (e.key === 'Enter') {
                    navigate();
                }
            });

            goBtn.addEventListener('click', navigate);
            backBtn.addEventListener('click', goBack);
            forwardBtn.addEventListener('click', goForward);
            refreshBtn.addEventListener('click', refresh);
            proxyToggle.addEventListener('click', toggleProxy);
            settingsBtn.addEventListener('click', openSettings);
            closeSettings.addEventListener('click', closeSettingsModal);
            saveSettings.addEventListener('click', saveSettingsModal);
            newTabBtn.addEventListener('click', addNewTab);

            // Webview events
            webview.addEventListener('loadstart', function() {
                loadingOverlay.classList.remove('hidden');
            });

            webview.addEventListener('load', function() {
                loadingOverlay.classList.add('hidden');
                updateNavigationButtons();
                updateUrlBar();
                addToHistory(webview.src);
            });

            // Functions
            function navigate() {
                let url = urlBar.value.trim();
                
                if (!url) return;
                
                // Add http:// if missing
                if (!url.startsWith('http://') && !url.startsWith('https://')) {
                    url = 'https://' + url;
                }
                
                if (proxyEnabled) {
                    loadingOverlay.classList.remove('hidden');
                    // Use the proxy server
                    const proxyUrl = currentProxyServer + url;
                    webview.src = proxyUrl;
                } else {
                    // Direct connection
                    webview.src = url;
                }
            }

            function goBack() {
                if (historyIndex > 0) {
                    historyIndex--;
                    webview.src = history[historyIndex];
                }
            }

            function goForward() {
                if (historyIndex < history.length - 1) {
                    historyIndex++;
                    webview.src = history[historyIndex];
                }
            }

            function refresh() {
                webview.contentWindow.location.reload();
            }

            function toggleProxy() {
                proxyEnabled = !proxyEnabled;
                if (proxyEnabled) {
                    proxyToggle.innerHTML = '<i class="fas fa-shield-alt mr-1"></i><span>Proxy: ON</span>';
                    proxyToggle.classList.remove('bg-gray-500');
                    proxyToggle.classList.add('bg-green-500');
                } else {
                    proxyToggle.innerHTML = '<i class="fas fa-shield-alt mr-1"></i><span>Proxy: OFF</span>';
                    proxyToggle.classList.remove('bg-green-500');
                    proxyToggle.classList.add('bg-gray-500');
                }
            }

            function updateNavigationButtons() {
                backBtn.disabled = historyIndex <= 0;
                forwardBtn.disabled = historyIndex >= history.length - 1;
            }

            function updateUrlBar() {
                // Extract the original URL from the proxy URL
                let displayUrl = webview.src;
                if (proxyEnabled && displayUrl.startsWith(currentProxyServer)) {
                    displayUrl = displayUrl.substring(currentProxyServer.length);
                }
                urlBar.value = displayUrl;
            }

            function addToHistory(url) {
                // Don't add duplicate consecutive entries
                if (history[historyIndex] === url) return;
                
                // If we're not at the end of history, truncate the future
                if (historyIndex < history.length - 1) {
                    history = history.slice(0, historyIndex + 1);
                }
                
                history.push(url);
                historyIndex = history.length - 1;
                updateNavigationButtons();
            }

            function openSettings() {
                proxyServerInput.value = currentProxyServer;
                enableJs.checked = javascriptEnabled;
                settingsModal.classList.remove('hidden');
            }

            function closeSettingsModal() {
                settingsModal.classList.add('hidden');
            }

            function saveSettingsModal() {
                currentProxyServer = proxyServerInput.value;
                // Ensure the proxy server URL ends with a slash
                if (!currentProxyServer.endsWith('/')) {
                    currentProxyServer += '/';
                }
                javascriptEnabled = enableJs.checked;
                saveSettings();
                closeSettingsModal();
            }

            function loadSettings() {
                const savedSettings = localStorage.getItem('proxyBrowserSettings');
                if (savedSettings) {
                    const settings = JSON.parse(savedSettings);
                    currentProxyServer = settings.proxyServer || currentProxyServer;
                    javascriptEnabled = settings.javascriptEnabled !== undefined ? settings.javascriptEnabled : true;
                }
            }

            function saveSettings() {
                const settings = {
                    proxyServer: currentProxyServer,
                    javascriptEnabled: javascriptEnabled
                };
                localStorage.setItem('proxyBrowserSettings', JSON.stringify(settings));
            }

            function addNewTab() {
                const tab = document.createElement('div');
                tab.className = 'tab px-4 py-2 flex items-center border-b-2 border-blue-500 text-blue-500';
                tab.innerHTML = `
                    <span>New Tab</span>
                    <button class="ml-2 text-gray-400 hover:text-gray-600">
                        <i class="fas fa-times"></i>
                    </button>
                `;
                
                // Remove active class from all tabs
                document.querySelectorAll('.tab').forEach(t => {
                    t.classList.remove('tab-active');
                });
                
                // Add new tab
                tabsContainer.appendChild(tab);
                
                // Clear the current tab
                webview.src = 'about:blank';
                urlBar.value = '';
                history = [];
                historyIndex = -1;
                updateNavigationButtons();
            }

            // Initial load
            webview.src = 'about:blank';
        });
    </script>
